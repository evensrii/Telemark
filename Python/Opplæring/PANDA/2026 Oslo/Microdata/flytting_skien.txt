// =====================================================================================
//  FLYTTING TIL OG FRA SKIEN - BEFOLKNINGEN SOM HELHET OG SEKUNDÆRBOSATTE FLYKTNINGER
// =====================================================================================
//
//  Formål:
//    Gruppe 1: Alle som har flyttet innenlands til eller fra Skien i analyseperioden.
//    Gruppe 2: Innflyttere til Skien med flukt som innvandringsgrunn og botid i Norge
//              på 10 år eller mindre ("sekundærbosatte"). Fordi de flytter innenlands
//              til Skien, har de per definisjon bodd i minst én annen norsk kommune.
//
//  Metode (øyeblikksbilder):
//    Bostedskommune og statuskode hentes per 1.1. hvert år. En person regnes som
//    innflytter i år t når hen er bosatt utenfor Skien 1.1.t og i Skien 1.1.(t+1).
//    Begge datoene krever statuskode 1 (bosatt). Dermed telles bare innenlands
//    flytting, og inn- og utvandring holdes utenfor.
//    Begrensning: flere flyttinger innenfor samme kalenderår fanges ikke opp
//    (f.eks. ut av Skien og tilbake igjen samme år).
//
//  Telleenhet:
//    Personer. Har en person flyttet til Skien flere ganger i perioden, brukes den
//    SISTE innflyttingen (flytteår, frakommune og kjennetegn). Tilsvarende for
//    utflytting. Antall flyttinger per år (hendelser) vises i egne tabeller.
//
//  Kommune- og fylkesnummer:
//    Alle kommunenummer harmoniseres til 2024-inndelingen (gjeldende nummer), med
//    tall som kodeformat. Skien = 4003, Telemark = 40, Midt-Telemark = 4020.
//    Fylke = kommunenummer / 100 (heltall), f.eks. 4003 -> 40 og 301 -> 3 (Oslo).
//    Omkodingen dekker koder fra 2019 og senere, så START kan settes til 2019.
//
//  Personvern (microdata.no):
//    - Tabeller får tilfeldig støy (+/- 5). Verdier 1-4 vises ikke.
//    - tabulate stoppes hvis over 50 % av cellene har under 5 observasjoner.
//      Krysstabeller for gruppe 2 holdes derfor grove.
//    - generate/replace/recode som berører færre enn 10 enheter avvises.
//      Harmonisering av kommunenummer gjøres derfor på HELE befolkningen,
//      før datasettet avgrenses til flytterne.
//
//  Overskrifter i output lages med textblock ... endblock (vises som Markdown).
//  textblocks er bevisst ikke lagt inne i for-løkker.
// =====================================================================================

- Telleenhet: personer (siste flytting i perioden)

// 3. IDENTIFISERE INN- OG UTFLYTTING (PER ÅR OG PER PERSON)
// -------------------------------------------------------------------------------------
// Variabler per person:
//   inn_<år> / ut_<år>     1 = flyttet til / fra Skien i året (hendelse)
//   siste_inn / siste_ut   flytteår for siste inn- og utflytting (0 = ingen)
//   ant_inn / ant_ut       antall inn- og utflyttinger i perioden

generate siste_inn = 0
generate siste_ut = 0
generate ant_inn = 0
generate ant_ut = 0
generate fra_kom = 0
generate til_kom = 0

for aar in $start : $slutt
  let neste = $aar + 1
  let kf = kom_ ++ $aar
  let kt = kom_ ++ $neste
  let sf = status_ ++ $aar
  let st = status_ ++ $neste
  let inn = inn_ ++ $aar
  let ut = ut_ ++ $aar

  // Innflytting: utenfor Skien 1.1.t, i Skien 1.1.(t+1), bosatt begge datoer

  replace $inn = 1 if $sf == '1' & $st == '1' & $kf != $skien & $kt == $skien

  // Utflytting: i Skien 1.1.t, utenfor Skien 1.1.(t+1), bosatt begge datoer

  replace $ut = 1 if $sf == '1' & $st == '1' & $kf == $skien & $kt != $skien

  replace fra_kom = $kf if $inn == 1
  replace ant_inn = ant_inn + 1 if $inn == 1

  replace siste_ut = $aar if $ut == 1
  replace til_kom = $kt if $ut == 1
  replace ant_ut = ant_ut + 1 if $ut == 1
end

import db/BEFOLKNING_INNGRUNN1 as inngrunn
import db/BEFOLKNING_FORSTDATO as forstdato
import db/BEFOLKNING_FODELAND as fodeland

// Alder i flytteåret (alder personen fyller i løpet av året)
generate alder_inn = siste_inn - int(foedsel / 100) if innflytter == 1
generate alder_ut = siste_ut - int(foedsel / 100) if utflytter == 1

define-labels aldgr 1 '0-15 år' 2 '16-19 år' 3 '20-24 år' 4 '25-29 år' 5 '30-39 år' 6 '40-49 år' 7 '50-66 år' 8 '67 år eller eldre'
generate aldgr_inn = alder_inn
recode aldgr_inn (0/15 = 1) (16/19 = 2) (20/24 = 3) (25/29 = 4) (30/39 = 5) (40/49 = 6) (50/66 = 7) (67/max = 8)
assign-labels aldgr_inn aldgr
generate aldgr_ut = alder_ut
recode aldgr_ut (0/15 = 1) (16/19 = 2) (20/24 = 3) (25/29 = 4) (30/39 = 5) (40/49 = 6) (50/66 = 7) (67/max = 8)
assign-labels aldgr_ut aldgr

// Botid i hele år: flytteår minus år for første opphold (ca., +/- 1 år).
// FORSTDATO har formatet ÅÅÅÅMMDD.
generate forst_aar = int(forstdato / 10000)
generate botid_inn = siste_inn - forst_aar if innflytter == 1
generate botid_ut = siste_ut - forst_aar if utflytter == 1

define-labels botidgr 1 '0-1 år' 2 '2-3 år' 3 '4-5 år' 4 '6-10 år' 5 'Over 10 år'
generate botidgr_inn = botid_inn
recode botidgr_inn (min/1 = 1) (2/3 = 2) (4/5 = 3) (6/10 = 4) (11/max = 5)
assign-labels botidgr_inn botidgr
generate botidgr_ut = botid_ut
recode botidgr_ut (min/1 = 1) (2/3 = 2) (4/5 = 3) (6/10 = 4) (11/max = 5)
assign-labels botidgr_ut botidgr

// Fødeland gruppert: Norge (landkode 000) eller utlandet
define-labels fland '1' 'Født i Norge' '2' 'Født i utlandet'
generate fland_grp = '1'
replace fland_grp = '2' if fodeland != '000'
assign-labels fland_grp fland


// -------------------------------------------------------------------------------------
// 5. HENTE KJENNETEGN SOM ENDRER SEG OVER TID (MÅLT I FLYTTEÅRET)
// -------------------------------------------------------------------------------------
// Hver variabel hentes for hvert flytteår og kobles til personens siste inn- og
// utflyttingsår. Referansedatoer:
//   ARBSTATUS_PERS_UTENFOR      16.11. (data til og med 2024-11-16)
//   REGSYS_ARB_ARBMARK_STATUS   16.11. (data til og med 2025-11-16)
//   REGSYS_ARB_YRKE_STYRK08     16.11. (data til og med 2025-11-16)
//   NUDB_BU                     1.10. (data til og med 2024-08-01)
// For flytteår som mangler data brukes siste tilgjengelige dato (se listene).
// JUSTER: listene skal ha ett element per flytteår fra START til SLUTT, i samme rekkefølge.
//
// 'X' betyr at variabelen ikke er hentet (personen flyttet ikke i den retningen).

generate arb_inn = 'X'
generate arb_ut = 'X'
generate amstat_inn = 'X'
generate amstat_ut = 'X'
generate yrke_inn = 'X'
generate yrke_ut = 'X'
generate bu_inn = 'X'
generate bu_ut = 'X'

for aar, d_arb, d_reg, d_utd in $start : $slutt, '2021-11-16' '2022-11-16' '2023-11-16' '2024-11-16' '2024-11-16', '2021-11-16' '2022-11-16' '2023-11-16' '2024-11-16' '2025-11-16', '2021-10-01' '2022-10-01' '2023-10-01' '2024-08-01' '2024-08-01'
  let a = arb_ ++ $aar
  let m = amstat_ ++ $aar
  let y = yrke_ ++ $aar
  let b = bu_ ++ $aar

  import db/ARBSTATUS_PERS_UTENFOR $d_arb as $a
  import db/REGSYS_ARB_ARBMARK_STATUS $d_reg as $m
  import db/REGSYS_ARB_YRKE_STYRK08 $d_reg as $y
  import db/NUDB_BU $d_utd as $b

  replace arb_inn = $a if siste_inn == $aar
  replace arb_ut = $a if siste_ut == $aar
  replace amstat_inn = $m if siste_inn == $aar
  replace amstat_ut = $m if siste_ut == $aar
  replace yrke_inn = $y if siste_inn == $aar
  replace yrke_ut = $y if siste_ut == $aar
  replace bu_inn = $b if siste_inn == $aar
  replace bu_ut = $b if siste_ut == $aar

  drop $a $m $y $b
end

// Arbeid, utdanning eller tiltak (gjelder personer 15 år og eldre)
define-labels arb '0' 'I arbeid, utdanning eller tiltak' '1' 'Utenfor arbeid, utdanning og tiltak' 'X' 'Ikke hentet'
assign-labels arb_inn arb
assign-labels arb_ut arb

// Arbeidsmarkedsstatus (hovedarbeidsforhold, 15-74 år)
define-labels amstat '0' 'Utenfor arbeidsstyrken' '1' 'Lønnstaker' '2' 'Selvstendig' '3' 'Helt ledig' '4' 'Ikke-lønnstaker' '9' 'Ikke-relevante lønnstakerforhold' 'X' 'Ikke hentet'
assign-labels amstat_inn amstat
assign-labels amstat_ut amstat

// Yrke: STYRK08 hovedgruppe (første siffer)
define-labels yrke1 '0' 'Militære yrker og uoppgitt' '1' 'Ledere' '2' 'Akademiske yrker' '3' 'Høyskoleyrker' '4' 'Kontor- og kundeserviceyrker' '5' 'Salgs- og serviceyrker' '6' 'Bønder, fiskere mv.' '7' 'Håndverkere' '8' 'Operatører, sjåfører mv.' '9' 'Renholdere, hjelpearbeidere mv.' 'X' 'Uoppgitt / ikke hentet'
generate yrke1_inn = substr(yrke_inn, 1, 1)
generate yrke1_ut = substr(yrke_ut, 1, 1)
assign-labels yrke1_inn yrke1
assign-labels yrke1_ut yrke1

// Utdanningsnivå: første siffer i NUS-koden, gruppert
//   0-2 grunnskole eller lavere, 3-5 videregående, 6 UH inntil 4 år, 7-8 UH over 4 år, 9 uoppgitt
define-labels utd '1' 'Grunnskole eller lavere' '2' 'Videregående' '3' 'Universitet/høyskole, kort' '4' 'Universitet/høyskole, lang' '9' 'Uoppgitt' 'X' 'Ikke hentet'
generate utd_inn = substr(bu_inn, 1, 1)
recode utd_inn ('0' '1' '2' = '1') ('3' '4' '5' = '2') ('6' = '3') ('7' '8' = '4')
assign-labels utd_inn utd
generate utd_ut = substr(bu_ut, 1, 1)
recode utd_ut ('0' '1' '2' = '1') ('3' '4' '5' = '2') ('6' = '3') ('7' '8' = '4')
assign-labels utd_ut utd


// -------------------------------------------------------------------------------------
// 6. DEFINERE GRUPPE 2: SEKUNDÆRBOSATTE FLYKTNINGER
// -------------------------------------------------------------------------------------
// Innflytter til Skien + innvandringsgrunn flukt + botid <= maks_botid i flytteåret.
generate gruppe2 = 0
replace gruppe2 = 1 if innflytter == 1 & inngrunn == 'FLU' & botid_inn <= $maks_botid

// Valgfritt: flyktninger med kort botid som har flyttet UT av Skien
generate gruppe2_ut = 0
replace gruppe2_ut = 1 if utflytter == 1 & inngrunn == 'FLU' & botid_ut <= $maks_botid


// =====================================================================================
// 7. RESULTATER GRUPPE 1: ALLE INN- OG UTFLYTTERE
// =====================================================================================

textblock
# Gruppe 1: Alle inn- og utflyttere
## 7.1 Hvor mange har flyttet?
Antall personer. En person kan være både innflytter og utflytter i perioden.
endblock

// --- 7.1 Hvor mange? ---------------------------------------------------------------
// Personer (siste flytting i perioden). Personer kan være både inn- og utflytter.
tabulate innflytter utflytter

textblock
**Antall flyttinger (hendelser) per år**: én tabell per år, innflytting (rader) mot utflytting (kolonner). Nettoflytting = inn - ut.
endblock

// Antall flyttinger (hendelser) per år. Nettoflytting = inn - ut.
for aar in $start : $slutt
  let inn = inn_ ++ $aar
  let ut = ut_ ++ $aar
  tabulate $inn $ut
end

// Antall inn- og utflyttinger per person i perioden
tabulate ant_inn if innflytter == 1
tabulate ant_ut if utflytter == 1

// --- 7.2 Når flyttet de? (år for siste flytting) ------------------------------------
textblock
## 7.2 Når flyttet de?
År for siste innflytting og siste utflytting.
endblock
tabulate siste_inn if innflytter == 1
tabulate siste_ut if utflytter == 1

// --- 7.3 Hvor flyttet de fra og til? ------------------------------------------------
textblock
## 7.3 Hvor flyttet de fra og til?
Fylke, fylke per år, de 25 vanligste kommunene og Telemark-kommunene for seg.
endblock
// Fylke
tabulate fra_fylke if innflytter == 1
tabulate til_fylke if utflytter == 1
tabulate siste_inn fra_fylke if innflytter == 1, rowpct
tabulate siste_ut til_fylke if utflytter == 1, rowpct

// Kommune (de 25 vanligste)
tabulate fra_kom if innflytter == 1, top(25)
tabulate til_kom if utflytter == 1, top(25)

// Telemark-kommunene for seg
tabulate fra_kom if innflytter == 1 & fra_fylke == 40
tabulate til_kom if utflytter == 1 & til_fylke == 40

// --- 7.4 Hva kjennetegner innflytterne? ---------------------------------------------
textblock
## 7.4 Hva kjennetegner innflytterne?
Kjønn, alder, innvandringsbakgrunn, fødeland, innvandringsgrunn, arbeid, arbeidsmarkedsstatus, utdanning og yrke (målt i flytteåret).
endblock
tabulate kjoenn if innflytter == 1
tabulate aldgr_inn if innflytter == 1
tabulate aldgr_inn kjoenn if innflytter == 1, colpct
tabulate invkat if innflytter == 1, missing
tabulate fland_grp if innflytter == 1
tabulate fodeland if innflytter == 1 & fland_grp == '2', top(20)
tabulate inngrunn if innflytter == 1 & invkat == 'B', missing
tabulate arb_inn if innflytter == 1 & alder_inn >= 15, missing
tabulate amstat_inn if innflytter == 1 & inrange(alder_inn, 15, 74), missing
tabulate utd_inn if innflytter == 1 & alder_inn >= 16, missing
tabulate yrke1_inn if innflytter == 1 & amstat_inn == '1'
tabulate fland_grp arb_inn if innflytter == 1 & alder_inn >= 15, rowpct

// --- 7.5 Hva kjennetegner utflytterne? ----------------------------------------------
textblock
## 7.5 Hva kjennetegner utflytterne?
Samme kjennetegn som for innflytterne, målt i året for siste utflytting.
endblock
tabulate kjoenn if utflytter == 1
tabulate aldgr_ut if utflytter == 1
tabulate aldgr_ut kjoenn if utflytter == 1, colpct
tabulate invkat if utflytter == 1, missing
tabulate fland_grp if utflytter == 1
tabulate fodeland if utflytter == 1 & fland_grp == '2', top(20)
tabulate inngrunn if utflytter == 1 & invkat == 'B', missing
tabulate arb_ut if utflytter == 1 & alder_ut >= 15, missing
tabulate amstat_ut if utflytter == 1 & inrange(alder_ut, 15, 74), missing
tabulate utd_ut if utflytter == 1 & alder_ut >= 16, missing
tabulate yrke1_ut if utflytter == 1 & amstat_ut == '1'
tabulate fland_grp arb_ut if utflytter == 1 & alder_ut >= 15, rowpct

// Alder som tall (gjennomsnitt, median)
summarize alder_inn alder_ut


// =====================================================================================
// 8. RESULTATER GRUPPE 2: SEKUNDÆRBOSATTE FLYKTNINGER (INNFLYTTING)
// =====================================================================================
// Gruppen er liten: hold tabellene grove. Hvis en tabell stoppes på grunn av mange
// små celler, fjern krysset eller bruk en grovere gruppering.

textblock
# Gruppe 2: Sekundærbosatte flyktninger
Innflyttere til Skien med **flukt** som innvandringsgrunn og **botid på 10 år eller mindre** i flytteåret.
Gruppen er liten, så tabellene holdes grove.

## 8.1 Hvor mange, og når flyttet de?
endblock

// --- 8.1 Hvor mange og når? ----------------------------------------------------------
tabulate gruppe2
tabulate siste_inn if gruppe2 == 1
// Andel av alle innflyttere som er sekundærbosatte flyktninger, per år
tabulate siste_inn gruppe2 if innflytter == 1, rowpct

// --- 8.2 Hvor flyttet de fra? -------------------------------------------------------
textblock
## 8.2 Hvor flyttet de fra?
endblock
tabulate fra_fylke if gruppe2 == 1
tabulate fra_kom if gruppe2 == 1, top(15)

// --- 8.3 Kjennetegn ----------------------------------------------------------------
textblock
## 8.3 Hva kjennetegner dem?
Kjønn, alder, botid, fødeland, arbeid, arbeidsmarkedsstatus, utdanning og yrke.
endblock
tabulate kjoenn if gruppe2 == 1
tabulate aldgr_inn if gruppe2 == 1
tabulate botidgr_inn if gruppe2 == 1
tabulate fodeland if gruppe2 == 1, top(15)
tabulate arb_inn if gruppe2 == 1 & alder_inn >= 15, missing
tabulate amstat_inn if gruppe2 == 1 & inrange(alder_inn, 15, 74), missing
tabulate utd_inn if gruppe2 == 1 & alder_inn >= 16, missing
tabulate yrke1_inn if gruppe2 == 1 & amstat_inn == '1'
tabulate botidgr_inn arb_inn if gruppe2 == 1 & alder_inn >= 15, rowpct
summarize alder_inn botid_inn if gruppe2 == 1

// --- 8.4 Valgfritt: flyktninger med kort botid som flyttet UT av Skien ----------------
textblock
## 8.4 Flyktninger med kort botid som flyttet ut av Skien
Valgfri del. Samme avgrensning som gruppe 2, men for utflytting.
endblock
tabulate gruppe2_ut
tabulate siste_ut if gruppe2_ut == 1
tabulate til_fylke if gruppe2_ut == 1
tabulate til_kom if gruppe2_ut == 1, top(15)
tabulate kjoenn if gruppe2_ut == 1
tabulate aldgr_ut if gruppe2_ut == 1
tabulate botidgr_ut if gruppe2_ut == 1
tabulate arb_ut if gruppe2_ut == 1 & alder_ut >= 15, missing


// =====================================================================================
// 9. OPPSUMMERING
// =====================================================================================
// Personer per gruppe:
//   innflytter == 1   -> alle innenlands innflyttere til Skien (gruppe 1, inn)
//   utflytter == 1    -> alle innenlands utflyttere fra Skien (gruppe 1, ut)
//   gruppe2 == 1      -> sekundærbosatte flyktninger (innflytting)
//   gruppe2_ut == 1   -> flyktninger med kort botid som flyttet ut (valgfritt)
textblock
# Oppsummering
- **innflytter = 1**: alle innenlands innflyttere til Skien (gruppe 1)
- **utflytter = 1**: alle innenlands utflyttere fra Skien (gruppe 1)
- **gruppe2 = 1**: sekundærbosatte flyktninger (innflytting)
- **gruppe2_ut = 1**: flyktninger med kort botid som flyttet ut (valgfritt)
endblock

tabulate innflytter gruppe2
tabulate utflytter gruppe2_ut
