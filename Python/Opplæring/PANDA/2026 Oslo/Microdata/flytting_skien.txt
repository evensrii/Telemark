//  FLYTTING TIL OG FRA SKIEN - BEFOLKNINGEN SOM HELHET OG SEKUNDÆRBOSATTE FLYKTNINGER

//
//  Formål:
//    Gruppe 1: Alle som har flyttet innenlands til eller fra Skien i analyseperioden.
//    Gruppe 2: Innflyttere til Skien med flukt som innvandringsgrunn og botid i Norge
//              på 10 år eller mindre ("sekundærbosatte"). Fordi de flytter innenlands
//              til Skien, har de per definisjon bodd i minst én annen norsk kommune.
//
//  Metode (øyeblikksbilder):
//    Bostedskommune og statuskode hentes per 1.1. hvert år. En person regnes som
//    innflytter i år t når hen er bosatt utenfor Skien 1.1.t og i Skien 1.1.(t+1).
//    Begge datoene krever statuskode 1 (bosatt). Dermed telles bare innenlands
//    flytting, og inn- og utvandring holdes utenfor.
//    Begrensning: flere flyttinger innenfor samme kalenderår fanges ikke opp
//    (f.eks. ut av Skien og tilbake igjen samme år).
//
//  Telleenhet:
//    Personer. Har en person flyttet til Skien flere ganger i perioden, brukes den
//    SISTE innflyttingen (flytteår, frakommune og kjennetegn). Tilsvarende for
//    utflytting. Antall flyttinger per år (hendelser) vises i egne tabeller.
//
//  Kommune- og fylkesnummer:
//    Alle kommunenummer harmoniseres til 2024-inndelingen (gjeldende nummer).
//    Skien = 4003, Telemark = 40, Midt-Telemark = 4020.
//
//  Personvern (microdata.no):
//    - Tabeller får tilfeldig støy (+/- 5). Verdier 1-4 vises ikke.
//    - tabulate stoppes hvis over 50 % av cellene har under 5 observasjoner.
//      Krysstabeller for gruppe 2 holdes derfor grove.
//    - generate/replace/recode som berører færre enn 10 enheter avvises.
//      Harmonisering av kommunenummer gjøres derfor på HELE befolkningen,
//      før datasettet avgrenses til flytterne.
//
//  Syntaks som bør testes første gang scriptet kjøres (ikke eksplisitt dokumentert):
//    - bindinger ($start osv.) i for-løkkenes verdiområde
//    - date_fmt($aar) med en binding som argument
//    Hvis noe av dette feiler: skriv inn tallene direkte i for-linjene.
//
//  Overskrifter i output lages med textblock ... endblock (vises som Markdown).
//  textblocks er bevisst ikke lagt inne i for-løkker.
// =====================================================================================

# Flytting til og fra Skien
Innenlands inn- og utflytting for **hele befolkningen** (gruppe 1) og for **sekundærbosatte flyktninger** (gruppe 2).

- Metode: bostedskommune 1.1. hvert år, sammenlignet år for år
- Telleenhet: personer (siste flytting i perioden)
- Kommune- og fylkesnummer: 2024-inndeling (Skien = 4003, Telemark = 40)
endblock

// -------------------------------------------------------------------------------------
// 3. IDENTIFISERE INN- OG UTFLYTTING (PER ÅR OG PER PERSON)
// -------------------------------------------------------------------------------------
// Variabler per person:
//   inn_<år> / ut_<år>     1 = flyttet til / fra Skien i året (hendelse)
//   siste_inn / siste_ut   flytteår for siste inn- og utflytting (0 = ingen)
//   ant_inn / ant_ut       antall inn- og utflyttinger i perioden
//   fra_kom / til_kom      kommunen personen kom fra / flyttet til ('0000' = ingen)

generate siste_inn = 0
generate siste_ut = 0
generate ant_inn = 0
generate ant_ut = 0
generate fra_kom = '0000'
generate til_kom = '0000'

for aar, neste in $start : $slutt, $start_pluss1 : $slutt_pluss1
  let kf = kom_ ++ $aar
  let kt = kom_ ++ $neste
  let sf = status_ ++ $aar
  let st = status_ ++ $neste
  let inn = inn_ ++ $aar
  let ut = ut_ ++ $aar

  // Innflytting: utenfor Skien 1.1.t, i Skien 1.1.(t+1), bosatt begge datoer
  generate $inn = 0
  replace $inn = 1 if $sf == '1' & $st == '1' & $kf != $skien & $kt == $skien

  // Utflytting: i Skien 1.1.t, utenfor Skien 1.1.(t+1), bosatt begge datoer
  generate $ut = 0
  replace $ut = 1 if $sf == '1' & $st == '1' & $kf == $skien & $kt != $skien

  // Løkken går i stigende årsrekkefølge, så den siste flyttingen overskriver tidligere

  replace fra_kom = $kf if $inn == 1
  replace ant_inn = ant_inn + 1 if $inn == 1

  replace siste_ut = $aar if $ut == 1
  replace til_kom = $kt if $ut == 1
  replace ant_ut = ant_ut + 1 if $ut == 1
end
