//
// ----------------------------------------------------------------------------

// ----------------------------------------------------------------------------
// 1) INTRO_FORST_BOSETTING_KOMMNR er i den offentlige variabelkatalogen oppgitt
//    med "gyldig dataperiode" 2022-12-31 til 2025-12-31. Her ANTAS det at
//    variabelen likevel dekker de eldste bosettingsårene vi trenger (x-9,
//    dvs. 2016 hvis x=2025), fordi den beskriver en fast, historisk hendelse
//    ved personen. Sjekk tabuleringen i STEG 2 nedenfor: hvis populasjonen
//    for de eldste bosettingsårene (2016-2017) blir urimelig lav/tom, er
//    antakelsen sannsynligvis feil, og en annen kilde til bosettingsår/
//    -kommune må brukes.
// 2) Flyktningbakgrunn identifiseres via BEFOLKNING_INNGRUNN1/BEFOLKNING_INVKAT.
//    Kodeverdien for "flyktning" i INNGRUNN1 (satt til '1' under i STEG 1) MÅ
//    dobbeltsjekkes mot kodeverket i microdata.no sitt variabelkatalog.
// 3) "Bosettingsår" er ikke en direkte variabel i katalogen, og avledes i
//    STEG 2 fra at INTRO_FORST_BOSETTING_KOMMNR er en STATUS-variabel (missing
//    før bosetting, har verdi fra og med bosettingsåret). Dette forutsetter at
//    variabelen aldri "hopper tilbake" til missing igjen - en rimelig antakelse
//    for en "første bosettingskommune"-variabel, men verifiser gjerne mot noen
//    kjente enkeltpersoner/kohorter i microdata.no.
// 4) Sentralitets- og kommunenummer-oppslagene (STEG 2 og STEG 4b) er meka-
//    nisk generert fra de vedlagte filene sentralitet.csv og
//    telemark_kommuner.csv, og er ikke i seg selv usikre - men de tabuleres
//    likevel med "missing" underveis, slik at evt. kommunenumre som ikke
//    fanges opp av oppslagene blir synlige.
// ============================================================================

// ----------------------------------------------------------------------------
// STEG 2c: Avled bosettingsår (se antakelse 3 i toppkommentaren)
// ----------------------------------------------------------------------------
// Vi sjekker INTRO_FORST_BOSETTING_KOMMNR på 6 tidspunkt: ett år FØR kohort-
// vinduet starter (aar0-5 = x-10, brukt for å utelukke de som var bosatt
// lenge før vinduet) t.o.m. selve observasjonsåret (aar0 = x-5). Antall av
// disse tidspunktene der variabelen ER missing forteller oss bosettingsåret.
generate antall_aar_missing = 0
for i in 0 : $antall_kohortaar
  let sjekk_aar = $aar0 - $antall_kohortaar + $i
  let dato_sjekk = date_fmt($sjekk_aar, 1, 1)
  import fdb/INTRO_FORST_BOSETTING_KOMMNR $dato_sjekk as kommune_sjekk ++ $sjekk_aar
  replace antall_aar_missing = antall_aar_missing + 1 if sysmiss(kommune_sjekk ++ $sjekk_aar)
end

generate bosettingsaar = ($aar0 - $antall_kohortaar) + antall_aar_missing
tabulate bosettingsaar, missing

// Behold bare de som IKKE allerede var bosatt før kohortvinduet startet
// (antall_aar_missing == 0 betyr bosatt før x-10, og skal utelates)
keep if antall_aar_missing >= 1

tabulate bosettingsaar, missing
