// ============================================================================

// Merk: bruker IKKE "keep if mottar_sosialhjelp == 1" - et permanent keep på en sensitiv variabel som sosialhjelp blir stoppet av avsløringskontrollen ("For få enheter i måldatasettet"), selv om det reelle antallet mottakere er stort nok. Løsningen er å betinge tabulate-kallene med "if mottar_sosialhjelp == 1" i stedet for å fysisk forkaste resten av populasjonen.
use skien_befolkning
import fdb/SOSHJLPFDT_MOTTAK $dato_t0 as sosialhjelp_mottak_raw
generate mottar_sosialhjelp = sosialhjelp_mottak_raw == '1'

tabulate befolkningsgruppe mottar_sosialhjelp, missing rowpct

// Av alle som mottar sosialhjelp i Skien i dag, hvor mange er flyktninger - og hvor mange er det ikke?
tabulate erflyktning if mottar_sosialhjelp == 1, missing freq cellpct
tabulate befolkningsgruppe if mottar_sosialhjelp == 1, missing freq cellpct

// Full INNVKAT-oppløsning (ikke bare de 3 samlekategoriene i innvandringskategori) - koder: A=Uten innvandringsbakgrunn, B=Innvandrer, C=Norskfødt med to utenlandsfødte foreldre, D=ikke i bruk, E=Utenlandsfødt med én norsk forelder, F=Norskfødt med én utenlandsfødt forelder, G=Født i utlandet av to norskfødte foreldre.
tabulate invkat if mottar_sosialhjelp == 1, missing freq cellpct

// "Norsk landbakgrunn" her definert bredere som alle INNVKAT-koder unntatt B (Innvandrer) - altså A/C/E/F/G. D er ikke i bruk.
generate norsk_landbakgrunn = invkat == 'A' | invkat == 'C' | invkat == 'E' | invkat == 'F' | invkat == 'G'
tabulate norsk_landbakgrunn if mottar_sosialhjelp == 1, missing freq cellpct
